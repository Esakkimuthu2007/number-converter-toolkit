// ============================================== // 1. Stack Data Structure (LIFO Principle) // Time Complexity: Push O(1), Pop O(1), Peek O(1) // ============================================== class Stack { constructor() { this.items = []; } // Push element onto top of stack push(element) { this.items.push(element); } // Remove and return top element (LIFO) pop() { if (this.isEmpty()) return null; return this.items.pop(); } // Inspect top element without removing peek() { if (this.isEmpty()) return null; return this.items[this.items.length - 1]; } // Check if stack is empty isEmpty() { return this.items.length === 0; } // Return number of elements size() { return this.items.length; } }
// ============================================== // 2. Decimal to Multi-Base Converter using Stack // Bases: 2 (Binary), 8 (Octal), 16 (Hexadecimal) // Time Complexity: O(log_base n) // ============================================== function convertDecimalToBase(decimalNum, base) { const stack = new Stack(); const digits = "0123456789ABCDEF"; let temp = decimalNum; if (temp === 0) { stack.push("0"); } else { // Push remainders to stack (least significant first) while (temp > 0) { const rem = temp % base; stack.push(digits[rem]); temp = Math.floor(temp / base); } } // Pop elements in LIFO order (most significant first) let result = ""; while (!stack.isEmpty()) { result += stack.pop(); } return result; }
// ============================================== // 3. String Toolkit Operations using Stack // Operations: Reversal, Palindrome, Word Reversal // Time Complexity: O(n) linear time // ============================================== // A. String Reversal function reverseString(str) { const s = new Stack(); for (let ch of str) s.push(ch); let reversed = ""; while (!s.isEmpty()) reversed += s.pop(); return reversed; } // B. Palindrome Checker function checkPalindrome(str) { const clean = str.toLowerCase().replace(/[^a-z0-9]/g, ""); const s = new Stack(); for (let ch of clean) s.push(ch); for (let i = 0; i < clean.length; i++) { if (clean[i] !== s.pop()) return false; } return true; } // C. Sentence Word Reversal function reverseWordOrder(sentence) { const words = sentence.trim().split(/\s+/); const s = new Stack(); words.forEach(w => s.push(w)); const rev = []; while (!s.isEmpty()) rev.push(s.pop()); return rev.join(" "); }
